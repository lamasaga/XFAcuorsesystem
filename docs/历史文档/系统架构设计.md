### 3.2 核心数据表设计

```sql
-- ========================================
-- 用户与认证
-- ========================================

-- 用户表
CREATE TABLE users (
    id              SERIAL PRIMARY KEY,
    username        VARCHAR(50) UNIQUE NOT NULL,
    password_hash   VARCHAR(255) NOT NULL,
    role            VARCHAR(20) NOT NULL,  -- admin, academic, teacher, student
    name            VARCHAR(50) NOT NULL,
    email           VARCHAR(100),
    phone           VARCHAR(20),
    avatar          VARCHAR(255),
    is_active       BOOLEAN DEFAULT true,
    last_login      TIMESTAMP,
    created_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 权限表 (可选，用于细粒度控制)
CREATE TABLE permissions (
    id              SERIAL PRIMARY KEY,
    code            VARCHAR(50) UNIQUE NOT NULL,  -- 如: schedule:edit, student:view
    name            VARCHAR(100) NOT NULL,
    module          VARCHAR(50)  -- 所属模块
);

-- 角色权限关联表
CREATE TABLE role_permissions (
    role            VARCHAR(20) NOT NULL,
    permission_id   INTEGER REFERENCES permissions(id),
    PRIMARY KEY (role, permission_id)
);

-- ========================================
-- 教师管理
-- ========================================

CREATE TABLE teachers (
    id              SERIAL PRIMARY KEY,
    user_id         INTEGER REFERENCES users(id),  -- 关联登录账号
    employee_no     VARCHAR(20) UNIQUE NOT NULL,   -- 工号
    name            VARCHAR(50) NOT NULL,
    gender          VARCHAR(10),
    phone           VARCHAR(20),
    email           VARCHAR(100),
    subjects        TEXT[],                        -- 任教科目数组
    shift_type      VARCHAR(20) DEFAULT 'fullday', -- morning/evening/fullday
    max_daily_hours INTEGER DEFAULT 6,
    available_slots JSONB,                         -- 可用时间槽 JSON
    hire_date       DATE,
    status          VARCHAR(20) DEFAULT 'active',  -- active/leave/retired
    created_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- ========================================
-- 学生管理
-- ========================================

CREATE TABLE students (
    id              SERIAL PRIMARY KEY,
    user_id         INTEGER REFERENCES users(id),  -- 可选，关联登录账号
    student_no      VARCHAR(20) UNIQUE NOT NULL,   -- 学号
    name            VARCHAR(50) NOT NULL,
    gender          VARCHAR(10),
    birth_date      DATE,
    id_card         VARCHAR(20),                   -- 身份证号
    class_id        INTEGER REFERENCES classes(id),
    enrollment_date DATE,                          -- 入学日期
    status          VARCHAR(20) DEFAULT 'active',  -- active/休学/毕业/转出
    address         TEXT,
    parent_name     VARCHAR(50),
    parent_phone    VARCHAR(20),
    emergency_contact VARCHAR(50),
    emergency_phone VARCHAR(20),
    remark          TEXT,
    created_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- ========================================
-- 班级管理
-- ========================================

CREATE TABLE classes (
    id              SERIAL PRIMARY KEY,
    name            VARCHAR(50) NOT NULL,          -- 如: 一年级1班
    grade           INTEGER NOT NULL,              -- 年级: 1-6
    class_no        INTEGER NOT NULL,              -- 班号: 1,2,3...
    head_teacher_id INTEGER REFERENCES teachers(id), -- 班主任
    room_id         INTEGER REFERENCES rooms(id),  -- 固定教室
    student_count   INTEGER DEFAULT 0,
    semester_id     INTEGER REFERENCES semesters(id),
    created_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    UNIQUE(grade, class_no, semester_id)
);

-- ========================================
-- 教室管理
-- ========================================

CREATE TABLE rooms (
    id              SERIAL PRIMARY KEY,
    room_no         VARCHAR(20) UNIQUE NOT NULL,   -- 教室编号
    name            VARCHAR(50),                   -- 教室名称
    building        VARCHAR(50),                   -- 所在楼栋
    floor           INTEGER,                       -- 楼层
    capacity        INTEGER,                       -- 容纳人数
    room_type       VARCHAR(30),                   -- 普通/多媒体/实验室/音乐/美术/体育
    equipment       TEXT[],                        -- 设备列表
    is_available    BOOLEAN DEFAULT true,
    remark          TEXT,
    created_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 教室预约表
CREATE TABLE room_reservations (
    id              SERIAL PRIMARY KEY,
    room_id         INTEGER REFERENCES rooms(id),
    date            DATE NOT NULL,
    periods         INTEGER[],                     -- 预约的节次
    purpose         VARCHAR(200),                  -- 用途
    applicant_id    INTEGER REFERENCES users(id),
    status          VARCHAR(20) DEFAULT 'pending', -- pending/approved/rejected
    created_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- ========================================
-- 科目与课程
-- ========================================

CREATE TABLE subjects (
    id              SERIAL PRIMARY KEY,
    code            VARCHAR(20) UNIQUE NOT NULL,
    name            VARCHAR(50) NOT NULL,
    category        VARCHAR(20),                   -- 文化课/艺术/体育/综合
    is_main         BOOLEAN DEFAULT false,         -- 是否主科
    required_room_type VARCHAR(30),                -- 所需教室类型
    difficulty      INTEGER DEFAULT 3,             -- 难度等级 1-5
    color           VARCHAR(10),                   -- 显示颜色
    created_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- ========================================
-- 学期与教学计划
-- ========================================

CREATE TABLE semesters (
    id              SERIAL PRIMARY KEY,
    name            VARCHAR(50) NOT NULL,          -- 如: 2025-2026学年第一学期
    school_year     VARCHAR(20),
    term            INTEGER,                       -- 1=第一学期, 2=第二学期
    start_date      DATE NOT NULL,
    end_date        DATE NOT NULL,
    is_current      BOOLEAN DEFAULT false,
    created_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 教学计划表
CREATE TABLE teaching_plans (
    id              SERIAL PRIMARY KEY,
    semester_id     INTEGER REFERENCES semesters(id),
    grade           INTEGER NOT NULL,
    subject_id      INTEGER REFERENCES subjects(id),
    weekly_hours    INTEGER NOT NULL,
    is_continuous   BOOLEAN DEFAULT false,
    continuous_count INTEGER DEFAULT 1,
    preferred_time  VARCHAR(20),                   -- morning/afternoon/any
    created_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    UNIQUE(semester_id, grade, subject_id)
);

-- ========================================
-- 排课相关
-- ========================================

-- 教学任务表 (谁教哪个班什么课)
CREATE TABLE teaching_tasks (
    id              SERIAL PRIMARY KEY,
    semester_id     INTEGER REFERENCES semesters(id),
    teacher_id      INTEGER REFERENCES teachers(id),
    class_id        INTEGER REFERENCES classes(id),
    subject_id      INTEGER REFERENCES subjects(id),
    weekly_hours    INTEGER NOT NULL,
    is_continuous   BOOLEAN DEFAULT false,
    continuous_count INTEGER DEFAULT 1,
    created_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    UNIQUE(semester_id, teacher_id, class_id, subject_id)
);

-- 课表条目表
CREATE TABLE schedule_entries (
    id              SERIAL PRIMARY KEY,
    task_id         INTEGER REFERENCES teaching_tasks(id),
    day_of_week     INTEGER NOT NULL,              -- 1-5 (周一到周五)
    period          INTEGER NOT NULL,              -- 1-8 (第几节)
    room_id         INTEGER REFERENCES rooms(id),
    is_locked       BOOLEAN DEFAULT false,         -- 是否锁定(手动固定)
    version         INTEGER DEFAULT 1,             -- 课表版本
    created_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    UNIQUE(task_id, day_of_week, period)
);

-- 约束配置表
CREATE TABLE constraints (
    id              SERIAL PRIMARY KEY,
    semester_id     INTEGER REFERENCES semesters(id),
    constraint_type VARCHAR(50) NOT NULL,          -- 约束类型
    config          JSONB NOT NULL,                -- 约束配置 JSON
    is_hard         BOOLEAN DEFAULT false,         -- 是否硬约束
    weight          INTEGER DEFAULT 5,             -- 软约束权重
    created_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- ========================================
-- 操作日志
-- ========================================

CREATE TABLE operation_logs (
    id              SERIAL PRIMARY KEY,
    user_id         INTEGER REFERENCES users(id),
    action          VARCHAR(50) NOT NULL,          -- 操作类型
    module          VARCHAR(50),                   -- 模块
    target_type     VARCHAR(50),                   -- 操作对象类型
    target_id       INTEGER,                       -- 操作对象ID
    detail          JSONB,                         -- 操作详情
    ip_address      VARCHAR(50),
    created_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
```
